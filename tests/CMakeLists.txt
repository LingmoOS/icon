set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Test)
function(add_lingmo_icon_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE Lingmo::Icon Qt6::Test)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
add_lingmo_icon_test(tst_IconTheme)
add_lingmo_icon_test(tst_IconLoader)
